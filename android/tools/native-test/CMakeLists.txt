cmake_minimum_required(VERSION 3.22)
project(qtest C CXX)
set(CMAKE_CXX_STANDARD 17)
set(WHISPER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/whisper.cpp)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+fp16+dotprod" CACHE STRING "" FORCE)
add_subdirectory(${WHISPER_DIR}/ggml ${CMAKE_BINARY_DIR}/ggml)
add_executable(qtest harness.c ${WHISPER_DIR}/src/whisper.cpp ${WHISPER_DIR}/src/aneforge/whisper-aneforge.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp/v2text_jni.c)
target_include_directories(qtest PRIVATE ${WHISPER_DIR}/include ${WHISPER_DIR}/src ${WHISPER_DIR}/ggml/include)
target_compile_definitions(qtest PRIVATE GGML_USE_CPU WHISPER_VERSION="vendored")
target_compile_options(qtest PRIVATE -O3 -march=armv8.2-a+fp16+dotprod)
target_link_libraries(qtest ggml -static-libstdc++ -static)
